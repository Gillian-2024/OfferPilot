```markdown
# OfferPilot 安全约定（五条硬规则）

> 这五条是硬门禁（`npm run check`）保护的对象。每条 = 一次已修的真实漏洞。

| # | 规则 | 保护对象 | 违反后果 |
|---|---|---|---|
| R1 | `auth.js` 的 `JWT_SECRET` 不得有兜底值（`process.env.JWT_SECRET \|\| '...'`）；缺则启动失败 | JWT 签名密钥 | 明文兜底=公开密钥，任何人可伪造任意用户 token |

| R5 | `index.js` 必须含 `rateLimit` 全站兜底 + `authLimiter` 挂载到 `/api/auth/login` + `/api/auth/register` | 限流 | 登录接口可被暴力破解 / 全站可被滥用 |

- 这些是**已落地**的硬约束。将来若架构演进（如新增独立前端域名），CORS 需白名单改造——那时先更新本表 + harness-check R2 正则，再改代码。
```

```js
// auth 单元测试：JWT 签发/验证、requireAuth 三态、无 JWT_SECRET 启动抛错。
// 注意：ESM import hoisting 使静态 import 在被测模块求值后才执行本文件语句，
// 故必须设 env 后动态 import（auth.js 顶层缺 JWT_SECRET 会 throw）。
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { dirname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

process.env.JWT_SECRET = 'cjc-test-secret-0123456789abcdef0123456789abcdef';

const __dirname = dirname(fileURLToPath(import.meta.url));
const SERVER_ROOT = resolve(__dirname, '..');
const { signToken, requireAuth } = await import('../src/middleware/auth.js');
const { TextEncoder } = await import('node:util');
const { jwtVerify } = await import('jose');

function mockRes() {
  const res = {};
  res.status = (code) => { res.statusCode = code; return res; };
  res.json = (body) => { res.body = body; return res; };
  return res;
}
// requireAuth 内部 jwtVerify 是 promise 链，调用后等一个 tick 再断言
function callMW(mw, req) {
  const res = mockRes();
  let nexted = false;
  mw(req, res, () => { nexted = true; });
  return new Promise((r) => setTimeout(r, 50)).then(() => ({ res, nexted }));
}

test('signToken 返回三段式 JWT', async () => {
  const token = await signToken('user-1', 'alice');
  assert.equal(token.split('.').length, 3);
});

test('signToken 签发 → jwtVerify 解出 userId 与 username', async () => {
  const token = await signToken('user-1', 'alice');
  const secret = new TextEncoder().encode(process.env.JWT_SECRET);
  const { payload } = await jwtVerify(token, secret);
  assert.equal(payload.sub, 'user-1');
  assert.equal(payload.username, 'alice');
});

test('requireAuth 无 Authorization 头 → 401 请先登录', async () => {
  const { res } = await callMW(requireAuth, { header: () => undefined });
  assert.equal(res.statusCode, 401);
  assert.equal(res.body.error, '请先登录');
});

test('requireAuth 无效 token → 401 登录已过期', async () => {
  const { res } = await callMW(requireAuth, { header: () => 'Bearer not.a.jwt' });
  assert.equal(res.statusCode, 401);
  assert.equal(res.body.error, '登录已过期，请重新登录');
});

test('requireAuth 有效 token → 设 req.userId 并 next()', async () => {
  const token = await signToken('user-1', 'alice');
  const req = { header: () => `Bearer ${token}` };
  const { res, nexted } = await callMW(requireAuth, req);
  assert.equal(res.statusCode, undefined);
  assert.equal(nexted, true);
  assert.equal(req.userId, 'user-1');
});

test('无 JWT_SECRET 时启动即抛错', () => {
  const r = spawnSync(process.execPath, ['--input-type=module', '-e',
    "await import('./src/middleware/auth.js')"],
    { cwd: SERVER_ROOT, env: { ...process.env, JWT_SECRET: '' }, encoding: 'utf8' });
  assert.notEqual(r.status, 0);
  assert.match(r.stderr, /缺少 JWT_SECRET/);
});
```

```js
#!/usr/bin/env node
// Harness 硬门禁：安全回归规则集（零依赖，fs.readFileSync + 正则扫源码）。
// 每条规则保护一次已修的安全洞（见 docs/harness/safety.md）；被破坏时 exit 1 阻止提交。
import { readFileSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = dirname(fileURLToPath(import.meta.url));
const ROOT = resolve(__dirname, '..');

const rules = [
  {
    id: 'R1',
    file: 'server/src/middleware/auth.js',
    mustNot: /process\.env\.JWT_SECRET\s*\|\|/,
    hint: 'JWT_SECRET 不得有兜底值（明文兜底=公开密钥，任何人可伪造任意用户 token）',
  },
  {
    id: 'R2',
    file: 'server/src/index.js',
    mustNot: /import\s+cors\s+from|require\(\s*['"]cors['"]\s*\)|\.use\(\s*cors\b/,
    hint: 'CORS 必须关闭（第三方站点跨域调本 API 会盗用用户 token）',
  },
  {
    id: 'R3',
    file: 'server/src/routes/history.js',
    mustNot: /INSERT\s+OR\s+REPLACE\s+INTO\s+history/i,
    hint: 'history upsert 必须带归属条件（裸 OR REPLACE 会覆盖他人记录）',
  },
  {
    id: 'R4',
    file: 'server/src/routes/chat.js',
    must: [/wrapUserContent/, /role\s*===\s*['"]user['"]/],
    hint: 'chat/rewrite 必须对 user 消息包 <user-content> 注入边界',
  },
  {
    id: 'R5',
    file: 'server/src/index.js',
    must: [/rateLimit\s*\(/, /authLimiter/, /app\.post\(\s*['"]\/api\/auth\/login['"]\s*,\s*authLimiter\s*\)/],
    hint: '必须含全站限流 + 登录/注册严限并挂载',
  },
];

let failed = 0;
for (const r of rules) {
  let src = '';
  try {
    src = readFileSync(resolve(ROOT, r.file), 'utf8');
  } catch {
    /* 文件缺失 → 视为违反，下面 must 检查会失败 */
  }
  const notViolated = !r.mustNot || !r.mustNot.test(src);
  const mustOk = !r.must || r.must.every((re) => re.test(src));
  if (notViolated && mustOk) {
    console.log(`✓ ${r.id}  ${r.file}  OK`);
  } else {
    failed += 1;
    console.error(`✗ ${r.id}  ${r.file}  违反\n    ${r.hint}`);
  }
}

if (failed > 0) {
  console.error(`\n安全回归硬门禁 FAIL：${failed} 条被破坏。修复后再提交。`);
  process.exit(1);
}
console.log('\n安全回归硬门禁全部通过 ✅');
```
